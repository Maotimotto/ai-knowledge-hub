from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel, Field
from transformers import AutoModelForCausalLM, AutoTokenizer

# ============ 全局模型管理 ============
class ModelManager:
    def __init__(self):
        self.model = None
        self.tokenizer = None
        self.device = "cuda" if torch.cuda.is_available() else "cpu"
        self.ready = False

    async def load(self, model_path: str):
        """异步加载模型，避免阻塞启动"""
        loop = asyncio.get_event_loop()
        await loop.run_in_executor(None, self._load_sync, model_path)
        self.ready = True

    def _load_sync(self, model_path: str):
        self.tokenizer = AutoTokenizer.from_pretrained(model_path)
        self.model = AutoModelForCausalLM.from_pretrained(
            model_path,
            torch_dtype=torch.float16,
            device_map="auto",
            trust_remote_code=True
        )
        self.model.eval()

    async def generate(self, prompt: str, **kwargs) -> str:
        if not self.ready:
            raise RuntimeError("模型尚未加载完成")
        loop = asyncio.get_event_loop()
        return await loop.run_in_executor(
            None, self._generate_sync, prompt, kwargs
        )

    def _generate_sync(self, prompt: str, params: dict) -> str:
        inputs = self.tokenizer(prompt, return_tensors="pt").to(self.device)
        with torch.no_grad():
            outputs = self.model.generate(
                **inputs,
                max_new_tokens=params.get("max_tokens", 512),
                temperature=params.get("temperature", 0.7),
                top_p=params.get("top_p", 0.9),
                do_sample=params.get("temperature", 0.7) > 0,
            )
        return self.tokenizer.decode(outputs[0][inputs["input_ids"].shape[1]:], skip_special_tokens=True)

model_manager = ModelManager()

# ============ 生命周期管理 ============

    start_time = time.time()

    # 构造 prompt
    prompt = build_prompt(request.messages)

    # 计算输入 token 数
    prompt_tokens = len(model_manager.tokenizer.encode(prompt))

    # 执行推理
    result = await model_manager.generate(
        prompt,
        max_tokens=request.max_tokens,
        temperature=request.temperature,
        top_p=request.top_p
    )

    completion_tokens = len(model_manager.tokenizer.encode(result))
    latency = time.time() - start_time

    # 记录指标
    record_metrics(request_id, prompt_tokens, completion_tokens, latency)

    return ChatResponse(
        id=request_id,
        created=int(time.time()),
        model=request.model,
        choices=[{
            "index": 0,
            "message": {"role": "assistant", "content": result},
            "finish_reason": "stop"
        }],
        usage=UsageInfo(
            prompt_tokens=prompt_tokens,
            completion_tokens=completion_tokens,
            total_tokens=prompt_tokens + completion_tokens
        )
    )

from fastapi.responses import StreamingResponse
import json

async def stream_generator(request: ChatRequest):
    """SSE 流式生成器"""
    request_id = f"chatcmpl-{uuid.uuid4().hex[:12]}"
    prompt = build_prompt(request.messages)

    inputs = model_manager.tokenizer(prompt, return_tensors="pt").to(model_manager.device)
    input_len = inputs["input_ids"].shape[1]

    # 使用 TextIteratorStreamer 实现流式输出
    from transformers import TextIteratorStreamer
    streamer = TextIteratorStreamer(model_manager.tokenizer, skip_prompt=True)

    generation_kwargs = {
        **inputs,
        "max_new_tokens": request.max_tokens,
        "temperature": request.temperature,
        "top_p": request.top_p,
        "streamer": streamer,
    }

    # 在后台线程执行生成
    import threading
    thread = threading.Thread(
        target=model_manager.model.generate,
        kwargs=generation_kwargs
    )
    thread.start()

    token_count = 0
    for text_chunk in streamer:
        token_count += 1
        chunk = {
            "id": request_id,
            "object": "chat.completion.chunk",
            "choices": [{
                "index": 0,
                "delta": {"content": text_chunk},
                "finish_reason": None
            }]
        }
        yield f"data: {json.dumps(chunk, ensure_ascii=False)}\n\n"

    # 发送结束标记
    final = {
        "id": request_id,
        "object": "chat.completion.chunk",
        "choices": [{"index": 0, "delta": {}, "finish_reason": "stop"}]
    }
    yield f"data: {json.dumps(final)}\n\n"
    yield "data: [DONE]\n\n"

@app.post("/v1/chat/completions/stream")
async def chat_completions_stream(request: ChatRequest):
    return StreamingResponse(
        stream_generator(request),
        media_type="text/event-stream"
    )

import time
from starlette.middleware.base import BaseHTTPMiddleware

class RequestLoggingMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request: Request, call_next):
        request_id = str(uuid.uuid4())[:8]
        start = time.time()

        # 注入 request_id
        request.state.request_id = request_id

        logger.info(f"[{request_id}] {request.method} {request.url.path}")

        try:
            response = await call_next(request)
        except Exception as e:
            logger.exception(f"[{request_id}] 请求处理异常")
            raise

        elapsed = time.time() - start
        logger.info(f"[{request_id}] 完成 | 状态={response.status_code} | 耗时={elapsed:.3f}s")
        response.headers["X-Request-ID"] = request_id
        response.headers["X-Process-Time"] = f"{elapsed:.3f}"
        return response

3.3 请求限流


import asyncio
from collections import defaultdict

class TokenBucketRateLimiter:
    """令牌桶限流器"""
    def __init__(self, rate: float, capacity: int):
        self.rate = rate          # 每秒填充速率
        self.capacity = capacity  # 桶容量
        self.buckets: dict[str, tuple[float, float]] = {}  # key -> (tokens, last_time)
        self._lock = asyncio.Lock()

    async def allow(self, key: str) -> bool:
        async with self._lock:
            now = time.time()
            if key not in self.buckets:
                self.buckets[key] = (self.capacity - 1, now)
                return True

            tokens, last_time = self.buckets[key]
            elapsed = now - last_time
            tokens = min(self.capacity, tokens + elapsed * self.rate)

            if tokens >= 1:
                self.buckets[key] = (tokens - 1, now)
                return True
            return False

rate_limiter = TokenBucketRateLimiter(rate=10, capacity=20)

@app.middleware("http")
async def rate_limit_middleware(request: Request, call_next):
    client_ip = request.client.host
    if not await rate_limiter.allow(client_ip):
        return JSONResponse(
            status_code=429,
            content={"error": {"message": "请求过于频繁", "code": 429}}
        )
    return await call_next(request)

import asyncio
import time
import random
from dataclasses import dataclass, field

@dataclass
class InstanceHealth:
    host: str
    port: int
    active_requests: int = 0
    avg_latency: float = 0.0
    error_count: int = 0
    last_health_check: float = 0.0
    is_healthy: bool = True
    gpu_memory_used: float = 0.0  # 0~1
    _latencies: list = field(default_factory=list)

    def update_latency(self, latency: float):
        self._latencies.append(latency)
        if len(self._latencies) > 100:
            self._latencies = self._latencies[-100:]
        self.avg_latency = sum(self._latencies) / len(self._latencies)

class SmartRouter:
    """智能请求路由器：综合负载、延迟、GPU 使用率进行调度"""

    def __init__(self, instances: list[tuple[str, int]]):
        self.instances = {
            f"{h}:{p}": InstanceHealth(host=h, port=p)
            for h, p in instances
        }
        self._lock = asyncio.Lock()

    async def select_instance(self) -> InstanceHealth:
        """选择最优实例"""
        healthy = [
            inst for inst in self.instances.values()
            if inst.is_healthy
        ]
        if not healthy:
            raise RuntimeError("没有可用的健康实例")

        # 加权评分：负载低 + 延迟低 + GPU 空闲 = 高分
        scores = []
        for inst in healthy:
            load_score = 1.0 / (1.0 + inst.active_requests)
            latency_score = 1.0 / (1.0 + inst.avg_latency)
            gpu_score = 1.0 - inst.gpu_memory_used
            score = 0.4 * load_score + 0.3 * latency_score + 0.3 * gpu_score
            scores.append((inst, score))

        # 加权随机选择（避免所有请求集中到同一实例）
        total = sum(s for _, s in scores)
        r = random.uniform(0, total)
        cumulative = 0
        for inst, score in scores:
            cumulative += score
            if r <= cumulative:
                return inst
        return scores[-1][0]

    async def health_check_loop(self, interval: float = 10.0):
        """周期性健康检查"""
        import aiohttp
        while True:
            async with aiohttp.ClientSession() as session:
                tasks = []
                for inst in self.instances.values():
                    tasks.append(self._check_one(session, inst))
                await asyncio.gather(*tasks, return_exceptions=True)
            await asyncio.sleep(interval)

    async def _check_one(self, session, inst: InstanceHealth):
        url = f"http://{inst.host}:{inst.port}/health"
        try:
            async with session.get(url, timeout=aiohttp.ClientTimeout(total=5)) as resp:
                if resp.status == 200:
                    data = await resp.json()
                    inst.is_healthy = True
                    inst.gpu_memory_used = data.get("gpu_memory_used_mb", 0) / 80000  # 归一化
                else:
                    inst.is_healthy = False
        except Exception:
            inst.error_count += 1
            if inst.error_count > 3:
                inst.is_healthy = False

from typing import Any

@dataclass
class QueuedRequest:
    prompt: str
    params: dict
    priority: int = 0       # 优先级，0=普通，1=高
    future: asyncio.Future = None
    created_at: float = 0.0

class RequestQueue:
    """优先级请求队列 + 超时管理"""

    def __init__(self, max_size: int = 100, default_timeout: float = 30.0):
        self.queue = asyncio.PriorityQueue(maxsize=max_size)
        self.default_timeout = default_timeout

    async def submit(self, prompt: str, params: dict, priority: int = 0, timeout: float = None) -> Any:
        future = asyncio.get_event_loop().create_future()
        req = QueuedRequest(
            prompt=prompt,
            params=params,
            priority=-priority,  # 取反使高优先级排前面
            future=future,
            created_at=time.time()
        )
        try:
            self.queue.put_nowait(req)
        except asyncio.QueueFull:
            raise HTTPException(429, "请求队列已满")

        try:
            return await asyncio.wait_for(future, timeout=timeout or self.default_timeout)
        except asyncio.TimeoutError:
            raise HTTPException(504, "推理超时")

    async def process_loop(self, router: SmartRouter):
        """持续从队列取出请求并分发"""
        while True:
            req = await self.queue.get()
            try:
                instance = await router.select_instance()
                async with instance:
                    instance.active_requests += 1
                    result = await self._call_instance(instance, req.prompt, req.params)
                    req.future.set_result(result)
            except Exception as e:
                req.future.set_exception(e)

# canary/traffic_router.py
import hashlib
import time
import random
from dataclasses import dataclass
from enum import Enum

class RoutingStrategy(Enum):
    PERCENTAGE = "percentage"    # 按百分比分流
    USER_COHORT = "user_cohort"  # 按用户群组分流
    HEADER_BASED = "header"      # 按请求头分流
    STICKY = "sticky"            # 粘性路由（同一用户始终命中同版本）

@dataclass
class ModelVersion:
    name: str
    endpoint: str
    weight: float          # 流量权重 0~100
    version: str
    is_canary: bool = False

class TrafficRouter:
    def __init__(self):
        self.versions: list[ModelVersion] = []
        self.strategy = RoutingStrategy.PERCENTAGE
        self._user_assignments: dict[str, str] = {}  # sticky 路由缓存

    def add_version(self, version: ModelVersion):
        self.versions.append(version)

    def route(self, request_context: dict) -> ModelVersion:
        """根据策略路由到对应版本"""
        if self.strategy == RoutingStrategy.PERCENTAGE:
            return self._route_by_percentage()
        elif self.strategy == RoutingStrategy.USER_COHORT:
            return self._route_by_cohort(request_context)
        elif self.strategy == RoutingStrategy.HEADER_BASED:
            return self._route_by_header(request_context)
        elif self.strategy == RoutingStrategy.STICKY:
            return self._route_sticky(request_context)

    def _route_by_percentage(self) -> ModelVersion:
        r = random.uniform(0, 100)
        cumulative = 0
        for v in self.versions:
            cumulative += v.weight
            if r <= cumulative:
                return v
        return self.versions[-1]

    def _route_by_cohort(self, ctx: dict) -> ModelVersion:
        """按用户 ID 哈希分流，保证同一用户始终进入同组"""
        user_id = ctx.get("user_id", "")
        h = int(hashlib.md5(user_id.encode()).hexdigest()[:8], 16)
        bucket = h % 100
        cumulative = 0
        for v in self.versions:
            cumulative += v.weight
            if bucket < cumulative:
                return v
        return self.versions[-1]

    def _route_by_header(self, ctx: dict) -> ModelVersion:
        """通过 X-Model-Version 头指定版本"""
        target = ctx.get("headers", {}).get("X-Model-Version")
        if target:
            for v in self.versions:
                if v.version == target:
                    return v
        return self._route_by_percentage()

    def _route_sticky(self, ctx: dict) -> ModelVersion:
        user_id = ctx.get("user_id", "")
        if user_id in self._user_assignments:
            version_name = self._user_assignments[user_id]
            for v in self.versions:
                if v.name == version_name:
                    return v
        # 首次分配
        chosen = self._route_by_percentage()
        self._user_assignments[user_id] = chosen.name
        return chosen

# canary/canary_controller.py
import asyncio
import logging
from datetime import datetime, timedelta
from dataclasses import dataclass, field

logger = logging.getLogger("canary-controller")

@dataclass
class CanaryConfig:
    """金丝雀发布配置"""
    new_version: str
    initial_weight: float = 5.0       # 初始流量比例
    increment: float = 10.0           # 每次增加的流量比例
    max_weight: float = 100.0         # 最大流量比例
    evaluation_interval: int = 300    # 评估间隔(秒)
    rollback_threshold: float = 0.05  # 错误率超过此值自动回滚
    latency_threshold: float = 2.0    # P95 延迟阈值(秒)
    min_requests: int = 100           # 最少请求数才开始评估

@dataclass
class CanaryMetrics:
    """金丝雀版本指标"""
    total_requests: int = 0
    error_count: int = 0
    latencies: list = field(default_factory=list)
    token_counts: list = field(default_factory=list)

    @property
    def error_rate(self) -> float:
        return self.error_count / max(self.total_requests, 1)

    @property
    def p95_latency(self) -> float:
        if not self.latencies:
            return 0
        sorted_lats = sorted(self.latencies)
        idx = int(len(sorted_lats) * 0.95)
        return sorted_lats[min(idx, len(sorted_lats) - 1)]

class CanaryController:
    def __init__(self, router: TrafficRouter, config: CanaryConfig):
        self.router = router
        self.config = config
        self.canary_metrics = CanaryMetrics()
        self.baseline_metrics = CanaryMetrics()
        self.state = "pending"  # pending, promoting, promoted, rollback
        self.started_at = None

    async def start(self):
        """启动金丝雀发布"""
        self.state = "promoting"
        self.started_at = datetime.now()

        # 添加新版本，初始低权重
        self.router.add_version(ModelVersion(
            name=f"canary-{self.config.new_version}",
            endpoint=f"http://llm-{self.config.new_version}:8000",
            weight=self.config.initial_weight,
            version=self.config.new_version,
            is_canary=True
        ))
        # 调整旧版本权重
        self._rebalance_weights()

        logger.info(f"金丝雀发布启动: {self.config.new_version}, 初始流量={self.config.initial_weight}%")

        # 启动评估循环
        await self._evaluation_loop()

    async def _evaluation_loop(self):
        """周期性评估金丝雀版本健康度"""
        while self.state == "promoting":
            await asyncio.sleep(self.config.evaluation_interval)

            if self.canary_metrics.total_requests < self.config.min_requests:
                logger.info(f"请求数不足({self.canary_metrics.total_requests}), 继续等待")
                continue

            # 评估指标
            should_rollback, reason = self._evaluate()
            if should_rollback:
                await self._rollback(reason)
                return

            # 判断是否可以升级流量
            canary_weight = self._get_canary_weight()
            if canary_weight >= self.config.max_weight:
                await self._promote()
                return

            # 增加金丝雀流量
            new_weight = min(canary_weight + self.config.increment, self.config.max_weight)
            self._update_canary_weight(new_weight)
            self._rebalance_weights()
            self._reset_metrics()

            logger.info(f"金丝雀流量提升至 {new_weight}%")

    def _evaluate(self) -> tuple[bool, str]:
        """评估金丝雀版本是否健康"""
        # 检查错误率
        if self.canary_metrics.error_rate > self.config.rollback_threshold:
            return True, f"错误率 {self.canary_metrics.error_rate:.2%} 超过阈值 {self.config.rollback_threshold:.2%}"

        # 检查延迟
        p95 = self.canary_metrics.p95_latency
        if p95 > self.config.latency_threshold:
            return True, f"P95延迟 {p95:.2f}s 超过阈值 {self.config.latency_threshold}s"

        # 对比基线（金丝雀不应比基线差太多）
        if self.baseline_metrics.latencies and self.canary_metrics.latencies:
            baseline_p50 = sorted(self.baseline_metrics.latencies)[len(self.baseline_metrics.latencies)//2]
            canary_p50 = sorted(self.canary_metrics.latencies)[len(self.canary_metrics.latencies)//2]
            if canary_p50 > baseline_p50 * 1.5:
                return True, f"金丝雀 P50延迟({canary_p50:.2f}s)比基线({baseline_p50:.2f}s)慢50%以上"

        return False, ""

    async def _rollback(self, reason: str):
        """回滚金丝雀版本"""
        self.state = "rollback"
        self.router.versions = [v for v in self.router.versions if not v.is_canary]
        self._rebalance_weights()
        logger.warning(f"金丝雀回滚: {reason}")
        # 发送告警
        await self._send_alert("rollback", reason)

    async def _promote(self):
        """金丝雀升级为正式版本"""
        self.state = "promoted"
        # 移除旧版本，金丝雀变为正式
        for v in self.router.versions:
            if v.is_canary:
                v.is_canary = False
                v.weight = 100.0
                v.name = v.version
        self.router.versions = [v for v in self.router.versions if not v.is_canary]
        logger.info(f"金丝雀发布成功: {self.config.new_version} 已全量")

    def _get_canary_weight(self) -> float:
        for v in self.router.versions:
            if v.is_canary:
                return v.weight
        return 0

    def _update_canary_weight(self, weight: float):
        for v in self.router.versions:
            if v.is_canary:
                v.weight = weight

    def _rebalance_weights(self):
        non_canary = [v for v in self.router.versions if not v.is_canary]
        canary = [v for v in self.router.versions if v.is_canary]
        if canary and non_canary:
            remaining = 100 - canary[0].weight
            for v in non_canary:
                v.weight = remaining / len(non_canary)

    def _reset_metrics(self):
        self.canary_metrics = CanaryMetrics()
        self.baseline_metrics = CanaryMetrics()

    async def _send_alert(self, event: str, detail: str):
        logger.warning(f"[ALERT] {event}: {detail}")

# ab_testing/experiment.py
import hashlib
import json
import time
from dataclasses import dataclass, field
from collections import defaultdict

@dataclass
class Experiment:
    name: str
    variants: list[dict]    # [{"name": "control", "model": "v1", "weight": 50}, ...]
    metrics: list[str]      # 要收集的指标 ["latency", "quality_score", "user_rating"]
    start_time: float = 0.0
    end_time: float = 0.0
    active: bool = True

class ABTestManager:
    def __init__(self):
        self.experiments: dict[str, Experiment] = {}
        self.results: dict[str, dict] = defaultdict(lambda: defaultdict(list))

    def create_experiment(self, exp: Experiment):
        self.experiments[exp.name] = exp
        exp.start_time = time.time()
        logger.info(f"实验创建: {exp.name}, 变体: {[v['name'] for v in exp.variants]}")

    def assign_variant(self, experiment_name: str, user_id: str) -> dict:
        """将用户分配到实验变体"""
        exp = self.experiments.get(experiment_name)
        if not exp or not exp.active:
            return exp.variants[0] if exp else {}

        # 基于用户ID + 实验名的哈希，保证同一用户在同一实验中始终同一变体
        key = f"{experiment_name}:{user_id}"
        h = int(hashlib.sha256(key.encode()).hexdigest()[:8], 16)
        bucket = h % 100

        cumulative = 0
        for variant in exp.variants:
            cumulative += variant.get("weight", 50)
            if bucket < cumulative:
                return variant
        return exp.variants[-1]

    def record_result(self, experiment_name: str, variant_name: str,
                      metric_name: str, value: float):
        """记录实验结果"""
        self.results[experiment_name][f"{variant_name}:{metric_name}"].append(value)

import asyncio
import logging
from typing import Optional

logger = logging.getLogger("shadow")

class ShadowController:
    """
    影子模式：新版本接收真实流量但不返回给用户，用于：
    1. 验证新版本稳定性
    2. 对比新旧版本输出质量
    3. 测量新版本性能指标
    """

    def __init__(self):
        self.shadow_enabled = False
        self.shadow_endpoint: Optional[str] = None
        self.comparison_results: list[dict] = []

    async def process_request(self, request_data: dict,
                               primary_handler, shadow_handler=None):
        """
        主流程处理 + 影子流量
        用户始终收到主版本的响应
        """
        # 正常处理主版本
        primary_result = await primary_handler(request_data)

        # 异步发送到影子版本（不阻塞主流程）
        if self.shadow_enabled and self.shadow_endpoint:
            asyncio.create_task(
                self._shadow_process(request_data, primary_result, shadow_handler)
            )

        return primary_result

    async def _shadow_process(self, request_data: dict,
                               primary_result: dict, shadow_handler):
        """异步处理影子流量"""
        try:
            shadow_start = time.time()
            shadow_result = await shadow_handler(self.shadow_endpoint, request_data)
            shadow_latency = time.time() - shadow_start

            # 对比分析
            comparison = {
                "timestamp": time.time(),
                "primary_tokens": primary_result.get("usage", {}).get("total_tokens", 0),
                "shadow_tokens": shadow_result.get("usage", {}).get("total_tokens", 0),
                "primary_latency": primary_result.get("_latency", 0),
                "shadow_latency": shadow_latency,
                "primary_output": primary_result["choices"][0]["message"]["content"][:200],
                "shadow_output": shadow_result["choices"][0]["message"]["content"][:200],
                "output_match": self._compare_outputs(
                    primary_result["choices"][0]["message"]["content"],
                    shadow_result["choices"][0]["message"]["content"]
                )
            }
            self.comparison_results.append(comparison)

            # 定期输出对比报告
            if len(self.comparison_results) % 100 == 0:
                self._log_report()

        except Exception as e:
            logger.error(f"影子处理失败: {e}")

    def _compare_outputs(self, primary: str, shadow: str) -> dict:
        """对比两个版本的输出"""
        return {
            "length_diff": len(shadow) - len(primary),
            "exact_match": primary.strip() == shadow.strip(),
        }

    def _log_report(self):
        """输出影子模式对比报告"""
        if not self.comparison_results:
            return
        recent = self.comparison_results[-100:]
        avg_primary_latency = sum(r["primary_latency"] for r in recent) / len(recent)
        avg_shadow_latency = sum(r["shadow_latency"] for r in recent) / len(recent)
        error_rate = sum(1 for r in recent if "error" in r) / len(recent)

        logger.info(
            f"影子模式报告 | 样本={len(recent)} | "
            f"主版本延迟={avg_primary_latency:.3f}s | "
            f"影子延迟={avg_shadow_latency:.3f}s | "
            f"影子错误率={error_rate:.2%}"
        )

    def enable_shadow(self, endpoint: str):
        self.shadow_enabled = True
        self.shadow_endpoint = endpoint
        logger.info(f"影子模式已启用: {endpoint}")

    def disable_shadow(self):
        self.shadow_enabled = False
        self.shadow_endpoint = None
        logger.info("影子模式已关闭")

    def get_comparison_report(self) -> dict:
        """获取完整对比报告"""
        if not self.comparison_results:
            return {"message": "暂无对比数据"}

        primary_lats = [r["primary_latency"] for r in self.comparison_results]
        shadow_lats = [r["shadow_latency"] for r in self.comparison_results]

        return {
            "total_comparisons": len(self.comparison_results),
            "primary": {
                "avg_latency": sum(primary_lats) / len(primary_lats),
                "p95_latency": sorted(primary_lats)[int(len(primary_lats) * 0.95)],
            },
            "shadow": {
                "avg_latency": sum(shadow_lats) / len(shadow_lats),
                "p95_latency": sorted(shadow_lats)[int(len(shadow_lats) * 0.95)],
            },
            "latency_change": f"{(sum(shadow_lats)/len(shadow_lats) - sum(primary_lats)/len(primary_lats)) / (sum(primary_lats)/len(primary_lats)) * 100:.1f}%"
        }

class ReleasePipeline:

        {"name": "shadow",       "duration": 1800, "canary_weight": 0},
        {"name": "canary_5",     "duration": 1800, "canary_weight": 5},
        {"name": "canary_20",    "duration": 3600, "canary_weight": 20},
        {"name": "canary_50",    "duration": 3600, "canary_weight": 50},
        {"name": "canary_100",   "duration": 1800, "canary_weight": 100},
    ]

    def __init__(self, controller: CanaryController, shadow: ShadowController):
        self.controller = controller
        self.shadow = shadow
        self.current_stage = 0
        self.state = "idle"

    async def run(self, new_version: str):
        self.state = "running"
        logger.info(f"开始发布流水线: {new_version}")

        for i, stage in enumerate(self.STAGES):
            self.current_stage = i
            logger.info(f"阶段 {i+1}/{len(self.STAGES)}: {stage['name']} | 流量={stage['canary_weight']}%")

            if stage["name"] == "shadow":
                self.shadow.enable_shadow(f"http://llm-{new_version}:8000")
                await asyncio.sleep(stage["duration"])
                report = self.shadow.get_comparison_report()
                if report.get("shadow", {}).get("p95_latency", 0) > 3.0:
                    logger.error("影子模式延迟过高，中止发布")
                    self.state = "failed"
                    return
                self.shadow.disable_shadow()
            else:
                self.controller._update_canary_weight(stage["canary_weight"])
                await asyncio.sleep(stage["duration"])

                # 检查指标
                if self.controller.canary_metrics.error_rate > 0.05:
                    logger.error(f"阶段 {stage['name']} 错误率过高，回滚")
                    await self.controller._rollback("错误率超标")
                    self.state = "failed"
                    return

        self.state = "completed"
        logger.info(f"发布完成: {new_version}")

class UsageRecord:

    model: str

    completion_tokens: int

    timestamp: float

class BillingManager:
    def __init__(self):
        self.records: list[UsageRecord] = []
        self.user_balance: dict[str, float] = {}  # 用户余额
        self.user_usage: dict[str, float] = defaultdict(float)  # 用户累计消费

    def calculate_cost(self, model: str, prompt_tokens: int, completion_tokens: int) -> float:
        """计算单次请求费用"""
        pricing = PRICING.get(model, PRICING["chat-7b"])
        cost = (prompt_tokens * pricing["input"] + completion_tokens * pricing["output"]) / 1000
        return round(cost, 6)

    def check_balance(self, user_id: str, estimated_cost: float) -> bool:
        """检查用户余额是否充足"""
        balance = self.user_balance.get(user_id, float("inf"))
        return balance >= estimated_cost

    def record_usage(self, user_id: str, api_key: str, model: str,
                     prompt_tokens: int, completion_tokens: int, request_id: str):
        """记录使用量并扣费"""
        cost = self.calculate_cost(model, prompt_tokens, completion_tokens)

        record = UsageRecord(
            user_id=user_id,
            api_key=api_key,
            model=model,
            prompt_tokens=prompt_tokens,
            completion_tokens=completion_tokens,
            cost=cost,
            timestamp=time.time(),
            request_id=request_id,
        )
        self.records.append(record)
        self.user_usage[user_id] += cost

        # 扣除余额
        if user_id in self.user_balance:
            self.user_balance[user_id] -= cost

        return record

    def get_user_report(self, user_id: str, start_time: float = 0, end_time: float = None) -> dict:
        """用户使用报告"""
        end_time = end_time or time.time()
        user_records = [
            r for r in self.records
            if r.user_id == user_id and start_time <= r.timestamp <= end_time
        ]

        if not user_records:
            return {"user_id": user_id, "total_cost": 0, "requests": 0}

        by_model = defaultdict(lambda: {"requests": 0, "cost": 0, "tokens": 0})
        for r in user_records:
            by_model[r.model]["requests"] += 1
            by_model[r.model]["cost"] += r.cost
            by_model[r.model]["tokens"] += r.prompt_tokens + r.completion_tokens

        return {
            "user_id": user_id,
            "period": f"{time.ctime(start_time)} ~ {time.ctime(end_time)}",
            "total_cost": round(sum(r.cost for r in user_records), 4),
            "total_requests": len(user_records),
            "total_tokens": sum(r.prompt_tokens + r.completion_tokens for r in user_records),
            "by_model": dict(by_model),
            "balance": self.user_balance.get(user_id, "unlimited"),

import hashlib
import json
import time
import numpy as np
from typing import Optional

class SemanticCache:
    """
    语义缓存：对语义相似的请求返回缓存结果，减少推理调用。
    使用 embedding 向量相似度匹配。
    """

    def __init__(self, similarity_threshold: float = 0.92, ttl: int = 3600):
        self.threshold = similarity_threshold
        self.ttl = ttl
        self.cache: list[dict] = []  # [{embedding, prompt, response, timestamp, hit_count}]

    def _compute_key(self, prompt: str, model: str, params: dict) -> str:
        """精确匹配的缓存键"""
        normalized = {
            "prompt": prompt.strip().lower(),
            "model": model,
            "temperature": params.get("temperature", 0.7),
            "max_tokens": params.get("max_tokens", 512),
        }
        return hashlib.sha256(json.dumps(normalized, sort_keys=True).encode()).hexdigest()

    async def get(self, prompt: str, model: str, params: dict,
                  embedding_model=None) -> Optional[dict]:
        """查询缓存，先精确匹配，再语义匹配"""
        key = self._compute_key(prompt, model, params)

        # 1. 精确匹配
        for entry in self.cache:
            if entry["key"] == key and time.time() - entry["timestamp"] < self.ttl:
                entry["hit_count"] += 1
                return entry["response"]

        # 2. 语义匹配
        if embedding_model:
            query_embedding = await embedding_model.encode(prompt)
            for entry in self.cache:
                if time.time() - entry["timestamp"] > self.ttl:
                    continue
                similarity = self._cosine_similarity(query_embedding, entry["embedding"])
                if similarity >= self.threshold:
                    entry["hit_count"] += 1
                    return entry["response"]

        return None

    async def put(self, prompt: str, model: str, params: dict,
                  response: dict, embedding_model=None):
        """写入缓存"""
        key = self._compute_key(prompt, model, params)
        embedding = None
        if embedding_model:
            embedding = await embedding_model.encode(prompt)

        self.cache.append({
            "key": key,
            "embedding": embedding,
            "prompt": prompt,
            "response": response,
            "timestamp": time.time(),
            "hit_count": 0,
        })

        # 过期清理
        self._evict_expired()

    def _cosine_similarity(self, a, b) -> float:
        a, b = np.array(a), np.array(b)
        return float(np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b)))

    def _evict_expired(self):
        now = time.time()
        self.cache = [e for e in self.cache if now - e["timestamp"] < self.ttl]

    def get_stats(self) -> dict:
        total_hits = sum(e["hit_count"] for e in self.cache)
        return {
            "entries": len(self.cache),
            "total_hits": total_hits,
            "avg_hits_per_entry": total_hits / max(len(self.cache), 1),
        }

# cache/redis_cache.py
import redis.asyncio as redis
import hashlib
import json

class RedisModelCache:
    """基于 Redis 的分布式推理缓存"""

    def __init__(self, redis_url: str = "redis://localhost:6379",
                 prefix: str = "llm_cache:", ttl: int = 7200):
        self.redis = redis.from_url(redis_url)
        self.prefix = prefix
        self.ttl = ttl

    async def get(self, prompt: str, model: str, **params) -> dict | None:
        key = self._make_key(prompt, model, params)
        data = await self.redis.get(key)
        if data:
            await self.redis.incr(f"{self.prefix}hits")
            return json.loads(data)
        await self.redis.incr(f"{self.prefix}misses")
        return None

    async def set(self, prompt: str, model: str, response: dict, **params):
        key = self._make_key(prompt, model, params)
        await self.redis.setex(key, self.ttl, json.dumps(response, ensure_ascii=False))

    def _make_key(self, prompt: str, model: str, params: dict) -> str:
        content = json.dumps({"p": prompt, "m": model, **params}, sort_keys=True)
        return f"{self.prefix}{hashlib.sha256(content.encode()).hexdigest()}"

    async def get_hit_rate(self) -> float:
        hits = int(await self.redis.get(f"{self.prefix}hits") or 0)
        misses = int(await self.redis.get(f"{self.prefix}misses") or 0)
        total = hits + misses
        return hits / total if total > 0 else 0

    async def get_stats(self) -> dict:
        keys = await self.redis.keys(f"{self.prefix}*")
        hits = int(await self.redis.get(f"{self.prefix}hits") or 0)
        misses = int(await self.redis.get(f"{self.prefix}misses") or 0)
        return {
            "cached_entries": len(keys),
            "cache_hits": hits,
            "cache_misses": misses,

"""
根据请求复杂度自动选择合适的模型：
- 简单问答 → 小模型（7B），低成本
- 复杂推理 → 大模型（70B），高准确率

import re
from typing import Literal

COMPLEXITY_PATTERNS = {
    "simple": [
        r"^(你好|hi|hello|谢谢|ok)",
        r"^(什么是|解释一下).{0,20}$",
        r"翻译.*为",
    ],
    "medium": [
        r"(分析|比较|总结|评价)",
        r"(写一篇|帮我写|生成)",
        r"代码.*实现",
    ],
    "complex": [
        r"(推理|证明|推导|数学)",
        r"(架构设计|系统设计)",
        r"(多步|链式|chain.?of.?thought)",
        r"代码.{20,}(优化|重构|debug)",
    ],
}

MODEL_TIERS = {
    "simple":  {"model": "chat-7b",  "cost_factor": 1.0},
    "medium":  {"model": "chat-13b", "cost_factor": 2.5},
    "complex": {"model": "chat-70b", "cost_factor": 10.0},
}

def estimate_complexity(prompt: str) -> Literal["simple", "medium", "complex"]:
    """估算 prompt 复杂度"""
    for level in ["complex", "medium", "simple"]:
        for pattern in COMPLEXITY_PATTERNS[level]:
            if re.search(pattern, prompt, re.IGNORECASE):
                return level
    # 按长度兜底
    if len(prompt) < 50:
        return "simple"
    elif len(prompt) < 500:
        return "medium"
    return "complex"

def select_model(prompt: str, user_tier: str = "standard",
                 force_model: str = None) -> dict:
    """选择最优模型"""
    if force_model:
        return {"model": force_model, "complexity": "forced"}

    complexity = estimate_complexity(prompt)

    # VIP 用户始终使用大模型
    if user_tier == "vip":
        complexity = "complex"

    tier_info = MODEL_TIERS[complexity]
    return {
        "model": tier_info["model"],
        "complexity": complexity,
        "cost_factor": tier_info["cost_factor"],
    }

# optimization/prompt_compression.py
"""
减少 prompt token 数量来降低成本：
1. 去除冗余空格和换行
2. 压缩系统提示词
3. 历史消息摘要
"""

def compress_prompt(messages: list[dict], max_history: int = 10) -> list[dict]:
    """压缩消息列表，减少 token 消耗"""
    compressed = []

    for msg in messages:
        if msg["role"] == "system":
            # 压缩系统提示：去除多余空行和空格
            content = " ".join(msg["content"].split())
            compressed.append({"role": "system", "content": content})
        else:
            compressed.append(msg)

    # 如果历史消息过多，做摘要
    if len(compressed) > max_history + 1:  # +1 for system message
        system = compressed[0] if compressed[0]["role"] == "system" else None
        history = compressed[1:] if system else compressed

        # 保留最近 N 条
        recent = history[-max_history:]

        # 对更早的消息做摘要
        older = history[:-max_history]
        summary = summarize_messages(older)

        result = []
        if system:
            result.append(system)
        result.append({"role": "system", "content": f"先前对话摘要: {summary}"})
        result.extend(recent)
        return result

    return compressed

def summarize_messages(messages: list[dict]) -> str:
    """简单摘要（生产中可用小模型生成）"
    topics = []
    for msg in messages:
        if msg["role"] == "user":
            topics.append(msg["content"][:50])
    return " | ".join(topics[:5]) + " ..."

def trim_context(prompt_tokens: list[int], max_tokens: int,
                 tokenizer) -> list[int]:
    """截断到最大 token 数，保留头尾"""
    if len(prompt_tokens) <= max_tokens:
        return prompt_tokens
    head = prompt_tokens[:max_tokens // 4]
    tail = prompt_tokens[-(max_tokens * 3 // 4):]
    return head + tail

# optimization/batch_inference.py
import asyncio
import time
from collections import deque

class BatchProcessor:
    """
    动态批处理：将多个请求合并为一个 batch 推理，提高 GPU 利用率。
    支持两种触发条件：batch_size 达到上限 或 超时。
    """

    def __init__(self, max_batch_size: int = 8, max_wait_ms: int = 50,
                 model_manager=None):
        self.max_batch_size = max_batch_size
        self.max_wait_ms = max_wait_ms
        self.model_manager = model_manager
        self.pending_queue = deque()
        self._processing = False

    async def submit(self, prompt: str, params: dict) -> str:
        """提交推理请求，返回结果"""
        future = asyncio.get_event_loop().create_future()
        self.pending_queue.append({
            "prompt": prompt,
            "params": params,
            "future": future,
            "submitted_at": time.time(),
        })

        # 触发批处理
        if not self._processing:
            asyncio.create_task(self._process_batch())

        return await future

    async def _process_batch(self):
        """收集请求并批量处理"""
        self._processing = True

        # 等待收集足够的请求或超时
        start = time.time()
        while len(self.pending_queue) < self.max_batch_size:
            elapsed_ms = (time.time() - start) * 1000
            if elapsed_ms >= self.max_wait_ms:
                break
            await asyncio.sleep(0.005)

        # 取出当前批次
        batch = []
        while self.pending_queue and len(batch) < self.max_batch_size:
            batch.append(self.pending_queue.popleft())

        if not batch:
            self._processing = False
            return

        # 批量推理
        try:
            prompts = [item["prompt"] for item in batch]
            results = await self._batch_generate(prompts, batch[0]["params"])

            for item, result in zip(batch, results):
                item["future"].set_result(result)
        except Exception as e:
            for item in batch:
                if not item["future"].done():
                    item["future"].set_exception(e)
        finally:
            self._processing = False

            # 如果等待队列中还有请求，继续处理
            if self.pending_queue:
                asyncio.create_task(self._process_batch())

    async def _batch_generate(self, prompts: list[str], params: dict) -> list[str]:
        """执行批量推理"""
        loop = asyncio.get_event_loop()
        return await loop.run_in_executor(None, self._batch_generate_sync, prompts, params)

    def _batch_generate_sync(self, prompts: list[str], params: dict) -> list[str]:
        tokenizer = self.model_manager.tokenizer
        model = self.model_manager.model
        device = self.model_manager.device

        # 批量编码
        inputs = tokenizer(prompts, return_tensors="pt", padding=True,
                          truncation=True, max_length=4096).to(device)

        with torch.no_grad():
            outputs = model.generate(
                **inputs,
                max_new_tokens=params.get("max_tokens", 512),
                temperature=params.get("temperature", 0.7),
                do_sample=params.get("temperature", 0.7) > 0,
            )

        # 解码结果
        results = []
        for i, output in enumerate(outputs):
            input_len = inputs["input_ids"][i].shape[0]
            text = tokenizer.decode(output[input_len:], skip_special_tokens=True)
            results.append(text)

        return results

# optimization/gpu_optimizer.py
import torch
import gc

class GPUOptimizer:
    """GPU 显存优化工具"""

    @staticmethod
    def enable_memory_efficient_attention():
        """启用 Flash Attention（需要硬件支持）"""
        if hasattr(torch.nn.functional, "scaled_dot_product_attention"):
            torch.backends.cuda.enable_flash_sdp(True)
            torch.backends.cuda.enable_mem_efficient_sdp(True)
            print("✅ Flash Attention 已启用")

    @staticmethod
    def optimize_model_memory(model):
        """优化模型显存占用"""
        # 1. 梯度检查点（推理时不需要，但模型加载时可设置）
        if hasattr(model, "gradient_checkpointing_enable"):
            model.gradient_checkpointing_enable()

        # 2. KV Cache 量化
        if hasattr(model.config, "use_cache"):
            model.config.use_cache = True

        return model

    @staticmethod
    def get_gpu_stats() -> dict:
        """获取 GPU 使用统计"""
        if not torch.cuda.is_available():
            return {"available": False}

        stats = {}
        for i in range(torch.cuda.device_count()):
            stats[f"gpu_{i}"] = {
                "name": torch.cuda.get_device_name(i),
                "memory_allocated_mb": round(torch.cuda.memory_allocated(i) / 1024**2, 1),
                "memory_reserved_mb": round(torch.cuda.memory_reserved(i) / 1024**2, 1),
                "memory_total_mb": round(torch.cuda.get_device_properties(i).total_mem / 1024**2, 1),
                "utilization_pct": round(
                    torch.cuda.memory_allocated(i) / torch.cuda.get_device_properties(i).total_mem * 100, 1
                ),
            }
        return stats

    @staticmethod
    def cleanup():
        """手动清理 GPU 缓存"""
        gc.collect()
        if torch.cuda.is_available():
            torch.cuda.empty_cache()
            torch.cuda.synchronize()

    @staticmethod
    def estimate_max_batch_size(model, tokenizer, prompt: str = "Hello",
                                 max_tokens: int = 512) -> int:
        """估算最大 batch size"""
        device = next(model.parameters()).device
        total_memory = torch.cuda.get_device_properties(device).total_mem
        allocated = torch.cuda.memory_allocated(device)
        available = total_memory * 0.85 - allocated  # 留 15% 余量

        # 单次推理显存估算
        inputs = tokenizer(prompt, return_tensors="pt").to(device)
        single_input_mem = inputs["input_ids"].element_size() * inputs["input_ids"].nelement()
        # 粗略估算：每个序列约占 single_input_mem * (max_tokens + input_len) * 2
        per_request = single_input_mem * (max_tokens + inputs["input_ids"].shape[1]) * 2

        max_batch = max(1, int(available / per_request))
        return min(max_batch, 32)  # 上限 32

import time
from collections import defaultdict

class CostReportGenerator:
    def __init__(self, billing_manager, cache):
        self.billing = billing_manager
        self.cache = cache

    async def generate_daily_report(self) -> dict:
        """生成每日成本报告"""
        now = time.time()
        day_ago = now - 86400

        daily_records = [
            r for r in self.billing.records
            if r.timestamp >= day_ago
        ]

        # 按用户聚合
        by_user = defaultdict(lambda: {"cost": 0, "requests": 0, "tokens": 0})
        by_model = defaultdict(lambda: {"cost": 0, "requests": 0, "tokens": 0})

        for r in daily_records:
            by_user[r.user_id]["cost"] += r.cost
            by_user[r.user_id]["requests"] += 1
            by_user[r.user_id]["tokens"] += r.prompt_tokens + r.completion_tokens

            by_model[r.model]["cost"] += r.cost
            by_model[r.model]["requests"] += 1
            by_model[r.model]["tokens"] += r.prompt_tokens + r.completion_tokens

        # 缓存节省
        cache_stats = await self.cache.get_stats() if self.cache else {}
        cache_hit_rate = float(cache_stats.get("hit_rate", "0%").replace("%", "")) / 100
        total_cost = sum(v["cost"] for v in by_user.values())
        estimated_savings = total_cost * cache_hit_rate / (1 - cache_hit_rate) if cache_hit_rate < 1 else 0

        return {
            "date": time.strftime("%Y-%m-%d"),
            "summary": {
                "total_cost_yuan": round(total_cost, 2),
                "total_requests": len(daily_records),
                "total_tokens": sum(r.prompt_tokens + r.completion_tokens for r in daily_records),
                "avg_cost_per_request": round(total_cost / max(len(daily_records), 1), 4),
                "cache_savings_yuan": round(estimated_savings, 2),
                "cache_hit_rate": cache_stats.get("hit_rate", "N/A"),
            },
            "by_user": {k: {**v, "cost": round(v["cost"], 4)} for k, v in
                        sorted(by_user.items(), key=lambda x: -x[1]["cost"])[:20]},
            "by_model": {k: {**v, "cost": round(v["cost"], 4)} for k, v in by_model.items()},
            "optimization_suggestions": self._generate_suggestions(by_model, cache_hit_rate),
        }

    def _generate_suggestions(self, by_model: dict, cache_hit_rate: float) -> list[str]:
        """生成优化建议"""
        suggestions = []

        total_cost = sum(v["cost"] for v in by_model.values())
        for model, data in by_model.items():
            if data["cost"] > total_cost * 0.5:
                suggestions.append(
                    f"模型 {model} 占总成本 {data['cost']/total_cost*100:.0f}%，"
                    f"考虑对简单请求使用更小模型"
                )

        if cache_hit_rate < 0.2:
            suggestions.append(
                f"缓存命中率仅 {cache_hit_rate*100:.0f}%，"
                f"建议调整缓存 TTL 或降低相似度阈值"
            )

        if not suggestions:
            suggestions.append("当前成本控制良好，无需调整")

        return suggestions
