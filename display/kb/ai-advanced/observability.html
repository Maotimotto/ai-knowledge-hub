LangSmith 追踪与评估

1. LangSmith 概述

LangSmith 是 LangChain 官方推出的 LLM 应用开发平台，提供追踪（Tracing）、评估（Evaluation）、

提示管理（Prompt Hub）和调试（Debugging）等核心能力。

核心能力

  • Tracing：记录每一步 LLM 调用、Chain 执行、Tool 调用的完整链路
  • Evaluation：构建评估数据集，自动化评测 LLM 输出质量
  • Prompt Hub：团队协作管理、版本化 Prompt 模板
  • Monitoring：生产环境实时监控延迟、成本、错误率

from langsmith import traceable
from openai import OpenAI

client = OpenAI()

@traceable(name="Chat Completion")
def chat_completion(user_message: str) -> str:
    response = client.chat.completions.create(
        model="gpt-4",
        messages=[{"role": "user", "content": user_message}]
    )
    return response.choices[0].message.content

@traceable(name="RAG Pipeline")
def rag_pipeline(question: str) -> str:
    # 检索阶段
    docs = retrieve_documents(question)  # 也会被追踪
    
    # 生成阶段
    context = "\n".join(docs)
    prompt = f"根据以下信息回答：\n{context}\n\n问题：{question}"
    return chat_completion(prompt)

# 调用时自动创建 trace
answer = rag_pipeline("LangSmith 的定价是什么？")

    # 模拟向量检索
    results = vector_store.similarity_search(query, k=top_k)
    return [doc.page_content for doc in results]

@traceable(
    name="LLM Generator",
    run_type="llm",
    tags=["generation", "gpt-4"]
)
def generate_answer(context: str, question: str) -> str:
    response = client.chat.completions.create(
        model="gpt-4",
        messages=[
            {"role": "system", "content": "基于上下文回答问题。"},
            {"role": "user", "content": f"上下文：{context}\n\n问题：{question}"}
        ],
        temperature=0.1
    )
    return response.choices[0].message.content

@traceable(name="RAG End-to-End")
def rag_e2e(question: str) -> str:
    docs = retrieve_documents(question)
    context = "\n---\n".join(docs)
    return generate_answer(context, question)

    4.2 运行评估

    
from langsmith.evaluation import evaluate

# 定义目标函数（你的 LLM 应用）
def target_function(inputs: dict) -> dict:
    result = rag_pipeline(inputs["question"])
    return {"answer": result}

# 定义评估器
def correctness_evaluator(run, example):
    """评估回答是否正确"""
    from openai import OpenAI
    client = OpenAI()
    
    response = client.chat.completions.create(
        model="gpt-4",
        messages=[{
            "role": "user",
            "content": f"""判断回答是否正确。
            问题：{example.inputs['question']}
            参考答案：{example.outputs['answer']}
            实际回答：{run.outputs['answer']}
            
            回答 CORRECT 或 INCORRECT，并简述原因。"""
        }]
    )
    
    is_correct = "CORRECT" in response.choices[0].message.content.upper()
    return {"key": "correctness", "score": 1.0 if is_correct else 0.0}

def conciseness_evaluator(run, example):
    """评估回答是否简洁"""
    answer = run.outputs.get("answer", "")
    score = 1.0 if len(answer) < 500 else 0.5 if len(answer) < 1000 else 0.0
    return {"key": "conciseness", "score": score}

# 运行评估
results = evaluate(
    target_function,
    data="Customer Support QA",
    evaluators=[correctness_evaluator, conciseness_evaluator],
    experiment_prefix="v2.1-test"
)

print(f"评估完成：{results}")

    3. 基本追踪

    3.1 装饰器方式（推荐）

    
from langfuse import Langfuse
from langfuse.decorators import observe, langfuse_context

langfuse = Langfuse()

@observe(as_type="generation")
def call_llm(prompt: str) -> str:
    from openai import OpenAI
    client = OpenAI()
    
    response = client.chat.completions.create(
        model="gpt-4",
        messages=[{"role": "user", "content": prompt}]
    )
    
    # 记录 token 用量和模型信息
    langfuse_context.update_current_observation(
        usage={
            "input": response.usage.prompt_tokens,
            "output": response.usage.completion_tokens,
            "total": response.usage.total_tokens
        },
        model="gpt-4",

    OpenTelemetry for LLM

    1. OpenTelemetry 概述

    OpenTelemetry（OTel）是 CNCF 的可观测性标准框架，提供统一的追踪（Traces）、指标（Metrics）、

    日志（Logs）采集协议。将其应用于 LLM 应用，可以实现与现有可观测性基础设施的无缝集成。

    为什么用 OTel 追踪 LLM？

  • 标准化：厂商无关的通用协议，避免锁定
  • 统一平台：LLM 追踪与后端服务追踪在同一平台展示
  • 生态丰富：支持 Grafana、Jaeger、Datadog、New Relic 等
  • 灵活控制：自定义采样、过滤、数据保留策略

from opentelemetry import trace
from opentelemetry.sdk.trace import TracerProvider
from opentelemetry.sdk.trace.export import BatchSpanProcessor
from opentelemetry.exporter.otlp.proto.grpc.trace_exporter import OTLPSpanExporter

provider = TracerProvider()
provider.add_span_processor(
    BatchSpanProcessor(OTLPSpanExporter(endpoint="http://localhost:4317"))
)
trace.set_tracer_provider(provider)
tracer = trace.get_tracer("rag-app")

def rag_pipeline(question: str) -> str:
    with tracer.start_as_current_span("rag.pipeline") as root_span:
        root_span.set_attribute("input.question", question)
        
        # Step 1: 文档检索
        with tracer.start_as_current_span("rag.retrieve") as retrieve_span:
            docs = retrieve_documents(question)
            retrieve_span.set_attribute("retrieval.doc_count", len(docs))
            retrieve_span.set_attribute("retrieval.method", "vector_search")
        
        # Step 2: 上下文构建
        with tracer.start_as_current_span("rag.context_build") as ctx_span:
            context = build_context(docs)
            ctx_span.set_attribute("context.token_count", count_tokens(context))
        
        # Step 3: LLM 生成
        with tracer.start_as_current_span("rag.generate") as gen_span:
            gen_span.set_attribute("gen_ai.system", "openai")
            gen_span.set_attribute("gen_ai.request.model", "gpt-4")
            
            answer = generate_answer(context, question)
            gen_span.set_attribute("output.answer_length", len(answer))
        
        root_span.set_attribute("output.answer", answer[:200])
        return answer

from opentelemetry import trace

tracer = trace.get_tracer("agent-app")

def agent_loop(user_input: str, max_iterations: int = 5) -> str:
    with tracer.start_as_current_span("agent.loop") as root:
        root.set_attribute("agent.max_iterations", max_iterations)
        messages = [{"role": "user", "content": user_input}]
        
        for i in range(max_iterations):
            with tracer.start_as_current_span(f"agent.step.{i}") as step:
                step.set_attribute("agent.iteration", i)
                
                # LLM 决策
                response = llm_call(messages)
                messages.append({"role": "assistant", "content": response})
                
                # 检查是否需要工具调用
                if needs_tool_call(response):
                    tool_name, tool_args = parse_tool_call(response)
                    step.set_attribute("agent.tool.name", tool_name)
                    
                    with tracer.start_as_current_span(f"agent.tool.{tool_name}") as tool:
                        tool_result = execute_tool(tool_name, tool_args)
                        tool.set_attribute("tool.result_length", len(str(tool_result)))
                    
                    messages.append({"role": "tool", "content": str(tool_result)})
                else:
                    step.set_attribute("agent.finished", True)
                    return response
        
        return "达到最大迭代次数"

        "thumbs_down": -1.0,
        "regenerate": -0.5,
        "copy": 0.3,

    3.3 自动降级策略

    
class LLMFallbackManager:
    """自动降级管理器"""
    
    def __init__(self):
        self.error_counts = {}
        self.circuit_breaker = {}
    
    def call_with_fallback(self, prompt: str, user_id: str) -> str:
        models = ["gpt-4", "gpt-4-turbo", "gpt-3.5-turbo"]
        
        for model in models:
            if self.is_circuit_open(model):
                continue
            
            try:
                result = self.call_model(model, prompt)
                self.record_success(model)
                return result
            except RateLimitError:
                self.record_failure(model)
                continue
            except TimeoutError:
                self.record_failure(model)
                if self.get_error_rate(model) > 0.5:
                    self.open_circuit(model)
                continue
            except Exception as e:
                self.record_failure(model)
                continue
        
        # 所有模型都失败，返回缓存或默认响应
        return self.get_cached_or_default(prompt)
    
    def is_circuit_open(self, model: str) -> bool:
        """断路器：连续失败过多时暂时跳过"""
        return self.circuit_breaker.get(model, {}).get("open", False)
    
    def get_error_rate(self, model: str) -> float:
        counts = self.error_counts.get(model, {"success": 0, "fail": 0})
        total = counts["success"] + counts["fail"]
        if total < 10:
            return 0.0
        return counts["fail"] / total

import numpy as np
from collections import deque

class AnomalyDetector:
    """基于统计的异常检测"""
    
    def __init__(self, window_size: int = 100, threshold: float = 3.0):
        self.window_size = window_size
        self.threshold = threshold  # 标准差倍数
        self.windows = {}
    
    def add_sample(self, metric_name: str, value: float) -> bool:
        if metric_name not in self.windows:
            self.windows[metric_name] = deque(maxlen=self.window_size)
        
        window = self.windows[metric_name]
        
        # 检测异常
        if len(window) >= 10:
            mean = np.mean(window)
            std = np.std(window)
            if std > 0 and abs(value - mean) > self.threshold * std:
                return True  # 异常！
        
        window.append(value)
        return False

# 使用示例
detector = AnomalyDetector(window_size=200, threshold=2.5)

def on_latency_sample(latency_ms: float, model: str):
    is_anomaly = detector.add_sample(f"latency.{model}", latency_ms)
    if is_anomaly:
        send_alert(f"异常延迟: {model} latency={latency_ms}ms")

import re

class DataSanitizer:
    """数据脱敏处理"""
    
    PATTERNS = {
        "email": r'\b[\w.-]+@[\w.-]+\.\w+\b',
        "phone": r'\b1[3-9]\d{9}\b',
        "id_card": r'\b\d{17}[\dXx]\b',
        "credit_card": r'\b\d{4}[\s-]?\d{4}[\s-]?\d{4}[\s-]?\d{4}\b',
    }
    
    @classmethod
    def sanitize(cls, text: str) -> str:
        for name, pattern in cls.PATTERNS.items():
            text = re.sub(pattern, f"[{name.upper()}_REDACTED]", text)
        return text

# 在发送到监控系统前脱敏
def safe_trace_data(prompt: str, response: str) -> dict:
    return {
        "prompt": DataSanitizer.sanitize(prompt),
        "response": DataSanitizer.sanitize(response),
        "prompt_length": len(prompt),
        "response_length": len(response),
    }
