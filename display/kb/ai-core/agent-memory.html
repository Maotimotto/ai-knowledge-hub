2. 滑动窗口记忆（Sliding Window）

保留最近 K 轮对话，丢弃超出窗口的历史消息。

2.1 实现原理

窗口大小 K=3 时的消息流：

时间 →
[msg1, msg2, msg3, msg4, msg5, msg6, msg7]
                                ┌───────────┐
                                │ 保留窗口   │
                                │ [msg5]     │
                                │ [msg6]     │
                                │ [msg7]     │
                                └───────────┘
[msg1, msg2, msg3, msg4] → 丢弃

2.2 代码实现

from collections import deque
from dataclasses import dataclass, field
from typing import List

@dataclass
class Message:
    role: str       # "user" | "assistant" | "system"
    content: str
    timestamp: float = 0.0

class SlidingWindowMemory:
    """滑动窗口短期记忆"""

    def __init__(self, window_size: int = 10, system_prompt: str = ""):
        self.window_size = window_size
        self.system_prompt = system_prompt
        self.messages: deque[Message] = deque(maxlen=window_size)

    def add_message(self, role: str, content: str):
        self.messages.append(Message(role=role, content=content))

    def get_context(self) -> List[dict]:
        context = []
        if self.system_prompt:
            context.append({"role": "system", "content": self.system_prompt})
        for msg in self.messages:
            context.append({"role": msg.role, "content": msg.content})
        return context

    def clear(self):
        self.messages.clear()

# 使用示例
memory = SlidingWindowMemory(window_size=6, system_prompt="你是一个友好的助手")
memory.add_message("user", "你好，我叫张三")
memory.add_message("assistant", "你好张三！有什么可以帮你的？")
memory.add_message("user", "我想学习Python")
context = memory.get_context()

3.3 手动摘要实现

from openai import OpenAI

client = OpenAI()

def summarize_conversation(messages: list, max_tokens: int = 200) -> str:
    """将对话历史压缩为摘要"""
    conversation_text = "\n".join(
        f"{m['role']}: {m['content']}" for m in messages
    )
    response = client.chat.completions.create(
        model="gpt-4o-mini",
        messages=[
            {"role": "system", "content": "请将以下对话压缩为简洁摘要，保留关键信息：用户名字、偏好、关键决定。"},
            {"role": "user", "content": conversation_text}
        ],
        max_tokens=max_tokens,
        temperature=0
    )
    return response.choices[0].message.content

class SummaryMemory:
    def __init__(self, llm, recent_k: int = 4):
        self.llm = llm
        self.recent_k = recent_k
        self.summary: str = ""
        self.messages: list = []

    def add(self, role: str, content: str):
        self.messages.append({"role": role, "content": content})
        if len(self.messages) > self.recent_k * 2:
            old_messages = self.messages[:-self.recent_k * 2]
            self.messages = self.messages[-self.recent_k * 2:]
            new_summary = summarize_conversation(old_messages)
            if self.summary:
                self.summary = f"{self.summary}\n{new_summary}"
            else:
                self.summary = new_summary

    def get_context(self) -> list:
        ctx = []
        if self.summary:
            ctx.append({"role": "system", "content": f"之前的对话摘要：{self.summary}"})
        ctx.extend(self.messages)
        return ctx

4.2 Token 计数与截断

import tiktoken

class TokenManagedMemory:
    def __init__(self, model: str = "gpt-4o", max_tokens: int = 4000):
        self.encoding = tiktoken.encoding_for_model(model)
        self.max_tokens = max_tokens
        self.messages: list = []
        self.system_prompt = ""

    def count_tokens(self, messages: list) -> int:
        total = 0
        for msg in messages:
            total += 4  # 每条消息的固定开销
            total += len(self.encoding.encode(msg["content"]))
        total += 2  # priming
        return total

    def add_message(self, role: str, content: str):
        self.messages.append({"role": role, "content": content})
        self._trim_to_fit()

    def _trim_to_fit(self):
        """从最早的消息开始移除，直到满足 token 限制"""
        while self.count_tokens(self.get_context()) > self.max_tokens and len(self.messages) > 1:
            self.messages.pop(0)

    def get_context(self) -> list:
        ctx = []
        if self.system_prompt:
            ctx.append({"role": "system", "content": self.system_prompt})
        ctx.extend(self.messages)
        return ctx

6.2 对话状态持久化

import json
import redis

class PersistentMemory:
    """基于 Redis 的持久化对话记忆"""

    def __init__(self, session_id: str, window_size: int = 10):
        self.session_id = session_id
        self.window_size = window_size
        self.redis = redis.Redis(host="localhost", port=6379, decode_responses=True)
        self.key = f"chat:session:{session_id}"

    def add_message(self, role: str, content: str):
        self.redis.rpush(self.key, json.dumps({"role": role, "content": content}))
        # 只保留最近的消息
        total = self.redis.llen(self.key)
        if total > self.window_size * 2:
            self.redis.lpop(self.key)
            self.redis.lpop(self.key)

    def get_context(self) -> list:
        raw = self.redis.lrange(self.key, 0, -1)
        return [json.loads(m) for m in raw]

    def clear(self):
        self.redis.delete(self.key)

    def get_summary(self) -> str:
        summary_key = f"{self.key}:summary"
        return self.redis.get(summary_key) or ""

    def save_summary(self, summary: str):
        summary_key = f"{self.key}:summary"
        self.redis.set(summary_key, summary, ex=86400 * 7)  # 7天过期

2. 向量存储记忆

2.1 核心原理

将文本通过 Embedding 模型转为高维向量，存入向量数据库，检索时通过余弦相似度匹配。

文本 → Embedding → [0.12, -0.34, 0.56, ...] → 向量数据库
                                                    │
用户查询 → Embedding → [0.11, -0.35, 0.55, ...] ──┘
                                                    │
                                          余弦相似度匹配
                                                    │
                                            返回最相关记忆

2.2 ChromaDB 实现

import chromadb
from chromadb.utils import embedding_functions

class VectorMemory:
    """基于 ChromaDB 的向量记忆"""

    def __init__(self, collection_name: str = "agent_memory"):
        self.client = chromadb.PersistentClient(path="./memory_db")
        self.ef = embedding_functions.OpenAIEmbeddingFunctions(
            model_name="text-embedding-3-small"
        )
        self.collection = self.client.get_or_create_collection(
            name=collection_name,
            embedding_function=self.ef
        )

    def add_memory(self, text: str, metadata: dict = None):
        """存储一条记忆"""
        doc_id = f"mem_{self.collection.count() + 1}"
        self.collection.add(
            documents=[text],
            metadatas=[metadata or {}],
            ids=[doc_id]
        )

    def search(self, query: str, top_k: int = 5) -> list:
        """语义检索记忆"""
        results = self.collection.query(
            query_texts=[query],
            n_results=top_k
        )
        return [
            {"text": doc, "score": score, "metadata": meta}
            for doc, score, meta in zip(
                results["documents"][0],
                results["distances"][0],
                results["metadatas"][0]
            )
        ]

    def get_context_for_query(self, query: str, top_k: int = 3) -> str:
        """为 LLM 构建记忆上下文"""
        memories = self.search(query, top_k)
        if not memories:
            return ""
        context = "相关记忆：\n"
        for m in memories:
            context += f"- {m['text']}\n"
        return context

# 使用示例

memory.add_memory("用户对机器学习感兴趣", {"category": "interest"})

# → 返回与"编程语言"语义最相关的记忆

3.2 NetworkX 实现

import networkx as nx
from typing import List, Tuple

class KnowledgeGraphMemory:
    """基于 NetworkX 的知识图谱记忆"""

    def __init__(self):
        self.graph = nx.DiGraph()

    def add_triple(self, subject: str, predicate: str, obj: str, metadata: dict = None):
        """添加三元组 (主语, 谓语, 宾语)"""
        self.graph.add_node(subject, type="entity")
        self.graph.add_node(obj, type="entity")
        self.graph.add_edge(subject, obj, relation=predicate, **(metadata or {}))

    def query_entity(self, entity: str) -> List[dict]:
        """查询实体相关的所有关系"""
        results = []
        if entity not in self.graph:
            return results
        for _, target, data in self.graph.edges(entity, data=True):
            results.append({
                "subject": entity,
                "predicate": data.get("relation", ""),
                "object": target
            })
        for source, _, data in self.graph.in_edges(entity, data=True):
            results.append({
                "subject": source,
                "predicate": data.get("relation", ""),
                "object": entity
            })
        return results

    def get_subgraph_context(self, entity: str, depth: int = 2) -> str:
        """获取实体的子图上下文"""
        if entity not in self.graph:
            return ""
        neighbors = nx.ego_graph(self.graph, entity, radius=depth)
        facts = []
        for u, v, data in neighbors.edges(data=True):
            facts.append(f"{u} --[{data.get('relation', '')}]--> {v}")
        return "\n".join(facts) if facts else ""

import sqlite3
import json
import time
from typing import Optional

class SQLiteMemory:
    """基于 SQLite 的结构化长期记忆"""

    def __init__(self, db_path: str = "memory.db"):
        self.conn = sqlite3.connect(db_path)
        self._init_db()

    def _init_db(self):
        self.conn.execute("""
            CREATE TABLE IF NOT EXISTS memories (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                session_id TEXT NOT NULL,
                memory_type TEXT NOT NULL,  -- 'fact', 'preference', 'event'
                content TEXT NOT NULL,
                metadata TEXT,
                importance REAL DEFAULT 0.5,
                created_at REAL,
                last_accessed REAL,
                access_count INTEGER DEFAULT 0
            )
        """)
        self.conn.execute("""
            CREATE INDEX IF NOT EXISTS idx_session ON memories(session_id)
        """)
        self.conn.commit()

    def store(self, session_id: str, memory_type: str, content: str,
              importance: float = 0.5, metadata: dict = None):
        now = time.time()
        self.conn.execute(
            "INSERT INTO memories (session_id, memory_type, content, metadata, importance, created_at, last_accessed) VALUES (?,?,?,?,?,?,?)",
            (session_id, memory_type, content, json.dumps(metadata or {}), importance, now, now)
        )
        self.conn.commit()

    def retrieve(self, session_id: str, memory_type: Optional[str] = None,
                 limit: int = 10) -> list:
        query = "SELECT * FROM memories WHERE session_id = ?"
        params = [session_id]
        if memory_type:
            query += " AND memory_type = ?"
            params.append(memory_type)
        query += " ORDER BY importance DESC, last_accessed DESC LIMIT ?"
        params.append(limit)

        rows = self.conn.execute(query, params).fetchall()
        return [
            {"id": r[0], "type": r[2], "content": r[3],
             "metadata": json.loads(r[4]), "importance": r[5]}
            for r in rows
        ]

import redis
import json

class RedisMemoryCache:
    """Redis 作为记忆的缓存层，加速热数据访问"""

    def __init__(self, host="localhost", port=6379):
        self.redis = redis.Redis(host=host, port=port, decode_responses=True)

    def cache_memory(self, key: str, data: dict, ttl_hours: int = 24):
        self.redis.setex(
            f"mem:{key}",
            timedelta(hours=ttl_hours),
            json.dumps(data, ensure_ascii=False)
        )

    def get_cached(self, key: str) -> dict:
        raw = self.redis.get(f"mem:{key}")
        return json.loads(raw) if raw else None

    def cache_search_results(self, query_hash: str, results: list, ttl_minutes: int = 30):
        """缓存向量检索结果，避免重复查询"""
        self.redis.setex(
            f"search:{query_hash}",
            timedelta(minutes=ttl_minutes),
            json.dumps(results, ensure_ascii=False)
        )

class HybridMemory:
    """混合存储：向量检索 + 结构化查询 + Redis 缓存"""

    def __init__(self):
        self.vector = VectorMemory("hybrid")
        self.sqlite = SQLiteMemory("hybrid.db")
        self.cache = RedisMemoryCache()

    def store(self, session_id: str, text: str, memory_type: str, importance: float = 0.5):
        # 同时存入向量库和结构化库
        self.vector.add_memory(text, {"session_id": session_id, "type": memory_type})
        self.sqlite.store(session_id, memory_type, text, importance)

    def retrieve(self, session_id: str, query: str, top_k: int = 5) -> list:
        # 1. 先查缓存
        cached = self.cache.get_cached(f"{session_id}:{query}")
        if cached:
            return cached

        # 2. 向量检索 + 结构化检索
        vector_results = self.vector.search(query, top_k)
        structured_results = self.sqlite.retrieve(session_id, limit=top_k)

        # 3. 融合结果（简单合并去重）
        seen = set()
        merged = []
        for r in vector_results:
            if r["text"] not in seen:
                seen.add(r["text"])
                merged.append(r)
        for r in structured_results:
            if r["content"] not in seen:
                seen.add(r["content"])
                merged.append({"text": r["content"], "metadata": r})

        # 4. 缓存结果
        self.cache.cache_memory(f"{session_id}:{query}", merged[:top_k])
        return merged[:top_k]

    def __init__(self, model: str = "gpt-4o"):
        self.model = model
        self.system_instructions = "你是一个有记忆的助手。"
        self.working_context = {
            "user_name": "",
            "preferences": "",
            "current_task": "",
        }
        self.message_queue = []  # FIFO
        self.max_queue_size = 10
        self.archival_store = []  # 简化版，实际用向量DB

    def _build_prompt(self) -> list:
        context_str = json.dumps(self.working_context, ensure_ascii=False)
        return [
            {"role": "system", "content": f"""{self.system_instructions}

## 工作上下文
{context_str}

## 可用函数
- core_memory_append(label, content): 更新工作上下文
- archival_memory_insert(content): 存入长期记忆
- archival_memory_search(query): 搜索长期记忆
"""},
            *self.message_queue
        ]

    def _execute_memory_action(self, action: dict):
        func = action.get("function")
        args = json.loads(action.get("arguments", "{}"))

        if func == "core_memory_append":
            label = args["label"]
            if label in self.working_context:
                self.working_context[label] += f"; {args['content']}"
        elif func == "core_memory_replace":
            self.working_context[args["label"]] = args["new_content"]
        elif func == "archival_memory_insert":
            self.archival_store.append(args["content"])

    def _manage_context(self):
        """FIFO 管理：上下文快满时摘要最旧的消息"""
        if len(self.message_queue) >= self.max_queue_size:
            old = self.message_queue[:2]
            summary = self._summarize(old)
            self.message_queue = self.message_queue[2:]
            self.archival_store.append(summary)

    def _summarize(self, messages: list) -> str:
        text = "\n".join(f"{m['role']}: {m['content']}" for m in messages)
        resp = client.chat.completions.create(
            model=self.model,
            messages=[
                {"role": "system", "content": "将对话压缩为一句话摘要"},
                {"role": "user", "content": text}
            ],
            max_tokens=100
        )
        return resp.choices[0].message.content

    def chat(self, user_message: str) -> str:
        self._manage_context()
        self.message_queue.append({"role": "user", "content": user_message})

        response = client.chat.completions.create(
            model=self.model,
            messages=self._build_prompt(),
            tools=self._get_tools()
        )

        msg = response.choices[0].message

        # 处理函数调用
        if msg.tool_calls:
            for tc in msg.tool_calls:
                self._execute_memory_action({
                    "function": tc.function.name,
                    "arguments": tc.function.arguments
                })
            self.message_queue.append({"role": "assistant", "content": msg.content or "(执行了记忆操作)"})
            return "已更新记忆。"

        self.message_queue.append({"role": "assistant", "content": msg.content})
        return msg.content

    def _get_tools(self) -> list:
        return [
            {"type": "function", "function": {
                "name": "core_memory_append",
                "description": "追加到工作上下文",
                "parameters": {"type": "object", "properties": {
                    "label": {"type": "string"},
                    "content": {"type": "string"}
                }, "required": ["label", "content"]}
            }},
            {"type": "function", "function": {
                "name": "archival_memory_insert",
                "description": "存入长期记忆",
                "parameters": {"type": "object", "properties": {
                    "content": {"type": "string"}
                }, "required": ["content"]}
            }}
        ]

2. 相关性评分（Relevance Scoring）

2.1 余弦相似度

import numpy as np
from openai import OpenAI

client = OpenAI()

def get_embedding(text: str) -> list:
    resp = client.embeddings.create(model="text-embedding-3-small", input=text)
    return resp.data[0].embedding

def cosine_similarity(a: list, b: list) -> float:
    a, b = np.array(a), np.array(b)
    return float(np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b)))

class RelevanceScorer:
    """记忆相关性评分器"""

    def __init__(self):
        self.embedding_cache = {}

    def score(self, query: str, memory_text: str) -> float:
        q_emb = self._get_cached_embedding(query)
        m_emb = self._get_cached_embedding(memory_text)
        return cosine_similarity(q_emb, m_emb)

    def _get_cached_embedding(self, text: str) -> list:
        if text not in self.embedding_cache:
            self.embedding_cache[text] = get_embedding(text)
        return self.embedding_cache[text]

    def rank(self, query: str, memories: list, top_k: int = 5) -> list:
        scored = [(m, self.score(query, m["text"])) for m in memories]
        scored.sort(key=lambda x: x[1], reverse=True)
        return scored[:top_k]

2.2 混合检索（BM25 + 向量）

from rank_bm25 import BM25Okapi
import jieba

class HybridRetriever:
    """BM25 + 向量 混合检索"""

    def __init__(self, alpha: float = 0.7):
        self.alpha = alpha  # 向量权重
        self.scorer = RelevanceScorer()

    def bm25_score(self, query: str, documents: list) -> list:
        tokenized = [list(jieba.cut(doc["text"])) for doc in documents]
        query_tokens = list(jieba.cut(query))
        bm25 = BM25Okapi(tokenized)
        scores = bm25.get_scores(query_tokens)
        # 归一化到 [0, 1]
        max_s = max(scores) if max(scores) > 0 else 1
        return [s / max_s for s in scores]

    def search(self, query: str, documents: list, top_k: int = 5) -> list:
        # BM25 得分
        bm25_scores = self.bm25_score(query, documents)

        # 向量得分
        vec_scores = [self.scorer.score(query, doc["text"]) for doc in documents]

        # 加权融合
        final = []
        for i, doc in enumerate(documents):
            combined = self.alpha * vec_scores[i] + (1 - self.alpha) * bm25_scores[i]
            final.append((doc, combined))

        final.sort(key=lambda x: x[1], reverse=True)
        return final[:top_k]

3.2 实现代码

import math
import time
from datetime import datetime

class TemporalDecay:
    """时间衰减计算"""

    @staticmethod
    def exponential(created_at: float, half_life_days: float = 7.0) -> float:
        """指数衰减：每 half_life_days 天衰减一半"""
        age_days = (time.time() - created_at) / 86400
        decay_rate = math.log(2) / half_life_days
        return math.exp(-decay_rate * age_days)

    @staticmethod
    def linear(created_at: float, max_days: float = 30.0) -> float:
        """线性衰减：max_days 天后衰减到 0"""
        age_days = (time.time() - created_at) / 86400
        return max(0.0, 1.0 - age_days / max_days)

    @staticmethod
    def logarithmic(created_at: float, scale: float = 10.0) -> float:
        """对数衰减：缓慢下降"""
        age_days = (time.time() - created_at) / 86400
        return 1.0 / (1.0 + math.log(1.0 + age_days / scale))

    @staticmethod
    def stepped(created_at: float, thresholds: list = None) -> float:
        """阶梯衰减"""
        if thresholds is None:
            thresholds = [(7, 1.0), (14, 0.7), (30, 0.4), (90, 0.1)]
        age_days = (time.time() - created_at) / 86400
        for max_days, score in thresholds:
            if age_days <= max_days:
                return score
        return 0.05

# 使用示例

one_week_ago = time.time() - 7 * 86400
one_month_ago = time.time() - 30 * 86400

print(f"指数衰减(1周前): {decay.exponential(one_week_ago):.3f}")  # 0.500
print(f"指数衰减(1月前): {decay.exponential(one_month_ago):.3f}")  # 0.024
print(f"线性衰减(1周前): {decay.linear(one_week_ago):.3f}")        # 0.767
print(f"对数衰减(1周前): {decay.logarithmic(one_week_ago):.3f}")   # 0.712

4. 重要性权重（Importance Weighting）

4.1 多维重要性评估

from dataclasses import dataclass

@dataclass
class MemoryItem:
    text: str
    created_at: float
    last_accessed: float
    access_count: int = 0
    importance: float = 0.5      # 人工标注或 LLM 评估
    emotional_valence: float = 0  # 情感强度
    memory_type: str = "fact"     # fact/preference/event/emotion

class ImportanceScorer:
    """多维重要性评估"""

    # 记忆类型权重
    TYPE_WEIGHTS = {
        "preference": 0.9,   # 用户偏好最重要
        "fact": 0.7,         # 事实次之
        "event": 0.5,        # 事件
        "emotion": 0.6,      # 情感
        "chat": 0.3,         # 普通对话
    }

    def score(self, memory: MemoryItem) -> float:
        """综合重要性得分 (0-1)"""
        # 1. 基础重要性
        base = memory.importance

        # 2. 类型权重
        type_w = self.TYPE_WEIGHTS.get(memory.memory_type, 0.5)

        # 3. 访问频率加成（频繁被访问 = 更重要）
        freq_bonus = min(memory.access_count / 10, 0.3)

        # 4. 情感加成
        emotion_bonus = abs(memory.emotional_valence) * 0.2

        # 5. 最近访问加成
        recency_bonus = 0.1 if (time.time() - memory.last_accessed) < 3600 else 0

        total = base * 0.4 + type_w * 0.3 + freq_bonus + emotion_bonus + recency_bonus
        return min(total, 1.0)

    def rank_by_importance(self, memories: list) -> list:
        scored = [(m, self.score(m)) for m in memories]
        scored.sort(key=lambda x: x[1], reverse=True)
        return scored

5.2 整合实现

class MemoryConsolidator:
    """记忆整合器"""

    def __init__(self, similarity_threshold: float = 0.85):
        self.threshold = similarity_threshold
        self.scorer = RelevanceScorer()
        self.importance = ImportanceScorer()

    def consolidate(self, memories: list) -> list:
        """执行记忆整合"""
        # 1. 去重：合并高度相似的记忆
        merged = self._merge_similar(memories)
        # 2. 遗忘：移除低重要性记忆
        filtered = self._forget_low_importance(merged)
        # 3. 更新重要性
        updated = self._update_importance(filtered)
        return updated

    def _merge_similar(self, memories: list) -> list:
        """合并语义相似的记忆"""
        if len(memories) <= 1:
            return memories

        merged = []
        used = set()

        for i, m1 in enumerate(memories):
            if i in used:
                continue
            group = [m1]
            for j, m2 in enumerate(memories):
                if j <= i or j in used:
                    continue
                sim = self.scorer.score(m1["text"], m2["text"])
                if sim > self.threshold:
                    group.append(m2)
                    used.add(j)

            if len(group) > 1:
                merged_text = self._merge_texts([g["text"] for g in group])
                merged.append({
                    "text": merged_text,
                    "importance": max(g.get("importance", 0.5) for g in group),
                    "created_at": min(g.get("created_at", time.time()) for g in group)
                })
            else:
                merged.append(m1)
            used.add(i)

        return merged

    def _merge_texts(self, texts: list) -> str:
        """用 LLM 合并相似文本"""
        response = client.chat.completions.create(
            model="gpt-4o-mini",
            messages=[{
                "role": "system",
                "content": "将以下相似信息合并为一条简洁完整的记忆，去除重复："
            }, {
                "role": "user",
                "content": "\n".join(f"- {t}" for t in texts)
            }],
            max_tokens=200,
            temperature=0
        )
        return response.choices[0].message.content

    def _forget_low_importance(self, memories: list, threshold: float = 0.2) -> list:
        """遗忘低重要性记忆"""
        return [m for m in memories if m.get("importance", 0.5) >= threshold]

    def _update_importance(self, memories: list) -> list:
        """更新记忆重要性（衰减旧记忆）"""
        for m in memories:
            decay = TemporalDecay.exponential(
                m.get("created_at", time.time()),
                half_life_days=14
            )
            m["adjusted_importance"] = m.get("importance", 0.5) * decay
        return memories

        self.retriever = HybridRetriever(alpha=0.7)
        self.decay = TemporalDecay()
        self.importance = ImportanceScorer()
        self.consolidator = MemoryConsolidator()

    def retrieve(self, query: str, memories: list, top_k: int = 5,
                 weights: dict = None) -> list:
        """
        综合检索：相关性 + 时间衰减 + 重要性

        weights: {"relevance": 0.5, "temporal": 0.2, "importance": 0.3}
        """
        if weights is None:
            weights = {"relevance": 0.5, "temporal": 0.2, "importance": 0.3}

        scored = []
        for mem in memories:
            # 相关性得分
            relevance = self.retriever.scorer.score(query, mem["text"])

            # 时间衰减
            temporal = self.decay.exponential(
                mem.get("created_at", time.time()),
                half_life_days=7
            )

            # 重要性
            importance = mem.get("importance", 0.5)

            # 加权综合
            final_score = (
                weights["relevance"] * relevance +
                weights["temporal"] * temporal +
                weights["importance"] * importance
            )

            scored.append({
                **mem,
                "scores": {
                    "relevance": round(relevance, 3),
                    "temporal": round(temporal, 3),
                    "importance": round(importance, 3),
                    "final": round(final_score, 3)
                }
            })

        scored.sort(key=lambda x: x["scores"]["final"], reverse=True)
        return scored[:top_k]
